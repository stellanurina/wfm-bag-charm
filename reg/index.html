/** * WFM Bag Charm Workshop ("Charmingly Yours") — Google Sheets Sync Bridge * Follows the exact 15-column layout (A:O) of "Sheet1" and "Check In": * A: Timestamp | B: Name | C: WhatsApp | D: Email | E: Charm Count | F: Charm Price * G: Inviting a Friend | H: Referred By | I: Invite Share URL * J: Payment Status | K: Payment Method | L: Birthday | M: Checked In | N: Check-In Time | O: Notes * * Fixes vs previous version: * - Birthday / Check-In Time / Timestamp cells that Sheets auto-converted to Dates are * returned as clean text ("12 Nov 1990", "10:15"), so the site can actually read them. * - Birthday, Check-In Time and WhatsApp are written as plain text so Sheets doesn't * turn "12 Nov" into a date or "0812..." into a number. * - An empty birthday / note / payment method from the site no longer wipes a real * value already in the sheet. * - Payment Method is returned as it really is in the sheet (no invented "Bank Transfer"). * - Rows are matched by the original registration name in the Invite Share URL too, so a * guest whose name was tidied up in the sheet (e.g. "Ann T." -> "Ann Tan") * updates her real row instead of creating a fake walk-in row. * - A registrant who is in Sheet1 but missing from "Check In" is copied from her real * Sheet1 row (real timestamp, referral, invite URL) instead of a made-up walk-in row. * * SECURITY — passcode: * Reading the guest list (GET) and check-in updates (POST action "checkin_update") only work * when the request carries the passcode stored in Project Settings → Script properties → * ACCESS_KEY. The passcode is never written in this code. If ACCESS_KEY isn't set, the guest * list stays locked. New signups from the wfm.stellas.digital landing page need no passcode * (they can only add a registration, never read or change anyone's details). */ const REG_SHEET_NAME = 'Sheet1'; const CHECKIN_SHEET_NAME = 'Check In'; const TZ = 'Asia/Jakarta'; function isAuthorized_(provided) { const key = PropertiesService.getScriptProperties().getProperty('ACCESS_KEY'); return Boolean(key) && String(provided || '') === String(key); } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function unauthorized_() { return jsonOut_({ status: 'unauthorized', message: 'Passcode required' }); } const COL = { TS: 1, NAME: 2, WA: 3, EMAIL: 4, COUNT: 5, PRICE: 6, INVITE: 7, REF: 8, URL: 9, PAY: 10, METHOD: 11, BDAY: 12, CHECKED: 13, TIME: 14, NOTES: 15 }; function normalizePhone_(val) { let digits = String(val || '').replace(/\D/g, ''); if (digits.startsWith('0')) digits = '62' + digits.slice(1); if (digits.startsWith('8')) digits = '62' + digits; return digits; } // Original registration name, taken from ".../invite.html?name=Ann%20T." function urlName_(url) { const m = String(url || '').match(/[?&]name=([^&#]*)/); if (!m) return ''; try { return decodeURIComponent(m[1].replace(/\+/g, ' ')).trim(); } catch (e) { return m[1].trim(); } } function isDate_(v) { return Object.prototype.toString.call(v) === '[object Date]' && !isNaN(v.getTime()); } function text_(v) { return String(v === null || v === undefined ? '' : v).trim(); } // Birthday cell -> "12 Nov 1990" / "12 Nov" (Sheets often stores these as real Dates) function birthdayText_(v) { if (!isDate_(v)) return text_(v); const y = Number(Utilities.formatDate(v, TZ, 'yyyy')); // "12 Nov" typed without a year gets the current year from Sheets — drop it return y >= 2021 ? Utilities.formatDate(v, TZ, 'd MMM') : Utilities.formatDate(v, TZ, 'd MMM yyyy'); } // Check-In Time cell -> "10:15" function timeText_(v) { return isDate_(v) ? Utilities.formatDate(v, TZ, 'HH:mm') : text_(v); } // Timestamp cell -> "2026-10-19 16:31" function stampText_(v) { return isDate_(v) ? Utilities.formatDate(v, TZ, 'yyyy-MM-dd HH:mm') : text_(v); } function getOrCreateCheckInSheet_(ss) { let sheet = ss.getSheetByName(CHECKIN_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CHECKIN_SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Timestamp', 'Name', 'WhatsApp', 'Email', 'Charm Count', 'Charm Price', 'Inviting a Friend', 'Referred By', 'Invite Share URL', 'Payment Status', 'Payment Method', 'Birthday', 'Checked In', 'Check-In Time', 'Notes' ]); } return sheet; } // Keep WhatsApp, Birthday and Check-In Time as plain text so Sheets doesn't convert them function setTextFormats_(sheet, row) { sheet.getRange(row, COL.WA).setNumberFormat('@'); sheet.getRange(row, COL.BDAY).setNumberFormat('@'); sheet.getRange(row, COL.TIME).setNumberFormat('@'); } /** * Finds the row (1-based) for a guest. Order: * 1. Name (sheet name OR original name in invite URL) + phone * 2. Name alone * 3. Phone alone, only if exactly one row has that phone (two guests can share one) */ function findRow_(data, name, phone) { const target = String(name || '').trim().toLowerCase(); const names = i => [text_(data[i][COL.NAME - 1]).toLowerCase(), urlName_(data[i][COL.URL - 1]).toLowerCase()]; if (target) { for (let i = 1; i < data.length; i++) { if (names(i).indexOf(target) !== -1 && (!phone || normalizePhone_(data[i][COL.WA - 1]) === phone)) return i + 1; } for (let i = 1; i < data.length; i++) { if (names(i).indexOf(target) !== -1) return i + 1; } } if (phone) { const hits = []; for (let i = 1; i < data.length; i++) { if (normalizePhone_(data[i][COL.WA - 1]) === phone) hits.push(i + 1); } if (hits.length === 1) return hits[0]; } return -1; } function rowToGuest_(r, rowNumber) { const charmCount = Number(r[COL.COUNT - 1]) || 2; const rawPrice = Number(String(r[COL.PRICE - 1] || '').replace(/[^0-9]/g, '')); const charmPrice = rawPrice >= 10000 ? rawPrice : (150000 + Math.max(0, charmCount - 2) * 50000); const payStatus = text_(r[COL.PAY - 1]).toLowerCase() === 'paid' ? 'Paid' : 'Unpaid'; const checked = r[COL.CHECKED - 1]; return { rowNumber: rowNumber, registeredAt: stampText_(r[COL.TS - 1]), name: text_(r[COL.NAME - 1]), registeredName: urlName_(r[COL.URL - 1]), whatsapp: text_(r[COL.WA - 1]), email: text_(r[COL.EMAIL - 1]), charmCount: charmCount, charmPrice: charmPrice, inviteStatus: text_(r[COL.INVITE - 1]) || 'No, flying solo', paymentStatus: payStatus, paymentMethod: payStatus === 'Paid' ? text_(r[COL.METHOD - 1]) : '', birthday: birthdayText_(r[COL.BDAY - 1]), checkedIn: text_(checked).toLowerCase() === 'cancelled' ? 'Cancelled' : (checked === true || text_(checked).toLowerCase() === 'yes' ? 'Yes' : 'No'), checkInTime: timeText_(r[COL.TIME - 1]), notes: text_(r[COL.NOTES - 1]) }; } // GET: Reads the 15-column "Check In" tab + any new signups from "Sheet1" function doGet(e) { if (!isAuthorized_(e && e.parameter && e.parameter.key)) return unauthorized_(); const ss = SpreadsheetApp.getActiveSpreadsheet(); const checkInSheet = getOrCreateCheckInSheet_(ss); const regSheet = ss.getSheetByName(REG_SHEET_NAME); const checkInData = checkInSheet.getDataRange().getValues(); const rows = []; const seenPhonesAndNames = new Set(); const seenUrls = new Set(); for (let i = 1; i < checkInData.length; i++) { const r = checkInData[i]; const name = text_(r[COL.NAME - 1]); if (!name || name.toLowerCase() === 'name') continue; seenPhonesAndNames.add(name.toLowerCase() + '|' + normalizePhone_(r[COL.WA - 1])); const inviteUrl = text_(r[COL.URL - 1]); if (inviteUrl) seenUrls.add(inviteUrl.toLowerCase()); rows.push(rowToGuest_(r, i + 1)); } // Also include any brand-new registrations on Sheet1 not yet in Check In if (regSheet) { const regData = regSheet.getDataRange().getValues(); for (let i = 1; i < regData.length; i++) { const r = regData[i]; const name = text_(r[COL.NAME - 1]); if (!name || name.toLowerCase() === 'name') continue; const inviteUrl = text_(r[COL.URL - 1]); if (seenPhonesAndNames.has(name.toLowerCase() + '|' + normalizePhone_(r[COL.WA - 1]))) continue; if (inviteUrl && seenUrls.has(inviteUrl.toLowerCase())) continue; rows.push(rowToGuest_(r, i + 1)); } } return ContentService.createTextOutput(JSON.stringify({ status: 'ok', syncedAt: new Date().toISOString(), rows: rows })).setMimeType(ContentService.MimeType.JSON); } // POST: Updates Check In columns E:F and J:O (matching your A:O headers) function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const payload = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Registrar Check-In / Payment / Birthday Update -> "Check In" tab if (payload.action === 'checkin_update') { if (!isAuthorized_(payload.key)) return unauthorized_(); const sheet = getOrCreateCheckInSheet_(ss); const data = sheet.getDataRange().getValues(); const targetPhone = normalizePhone_(payload.whatsapp); let rowIndex = findRow_(data, payload.name, targetPhone); // Not in "Check In" yet? Copy her REAL registration row (A:I) from Sheet1 first. if (rowIndex === -1) { const regSheet = ss.getSheetByName(REG_SHEET_NAME); if (regSheet) { const regData = regSheet.getDataRange().getValues(); const regRow = findRow_(regData, payload.name, targetPhone); if (regRow > 0) { const src = regData[regRow - 1].slice(0, 9); sheet.appendRow(src.concat(['Unpaid', '', '', 'No', '', ''])); rowIndex = sheet.getLastRow(); sheet.getRange(rowIndex, COL.WA).setNumberFormat('@').setValue(text_(src[COL.WA - 1])); } } } const charmCount = Number(payload.charmCount) || 2; const charmPrice = Number(payload.charmPrice) || (150000 + Math.max(0, charmCount - 2) * 50000); const isPaid = String(payload.paymentStatus || '').toLowerCase() === 'paid'; const isCheckedIn = payload.checkedIn === true || String(payload.checkedIn).toLowerCase() === 'yes'; const incomingBirthday = text_(payload.birthday); const incomingNotes = text_(payload.notes); const incomingMethod = text_(payload.paymentMethod); if (rowIndex > 0) { const current = sheet.getRange(rowIndex, 1, 1, 15).getValues()[0]; // Never wipe real sheet data with an empty value from the site const birthday = incomingBirthday || birthdayText_(current[COL.BDAY - 1]); // The page sends notesFinal: true once it has the sheet's notes loaded, so it can clear // a note (e.g. the "collect at desk" reminder once she pays). Older pages can't wipe notes. const notes = payload.notesFinal === true ? incomingNotes : (incomingNotes || text_(current[COL.NOTES - 1])); const payMethod = isPaid ? (incomingMethod || text_(current[COL.METHOD - 1])) : ''; setTextFormats_(sheet, rowIndex); sheet.getRange(rowIndex, COL.COUNT, 1, 2).setValues([[charmCount, charmPrice]]); sheet.getRange(rowIndex, COL.PAY, 1, 6).setValues([[ isPaid ? 'Paid' : 'Unpaid', payMethod, birthday, payload.cancelled === true ? 'Cancelled' : (isCheckedIn ? 'Yes' : 'No'), isCheckedIn ? text_(payload.checkInTime) : '', notes ]]); } else { // Genuine walk-in (not in Sheet1 or Check In) -> append full 15-column row (A:O) sheet.appendRow([ new Date(), payload.name || '', '', payload.email || '', charmCount, charmPrice, payload.inviteStatus || 'Walk-In', '', '', isPaid ? 'Paid' : 'Unpaid', isPaid ? incomingMethod : '', '', isCheckedIn ? 'Yes' : 'No', '', incomingNotes ]); const newRow = sheet.getLastRow(); setTextFormats_(sheet, newRow); sheet.getRange(newRow, COL.WA).setValue(text_(payload.whatsapp)); sheet.getRange(newRow, COL.BDAY).setValue(incomingBirthday); sheet.getRange(newRow, COL.TIME).setValue(isCheckedIn ? text_(payload.checkInTime) : ''); } return ContentService.createTextOutput(JSON.stringify({ status: 'ok', target: CHECKIN_SHEET_NAME, row: rowIndex })) .setMimeType(ContentService.MimeType.JSON); } // 2. Main Landing Page Registration (wfm.stellas.digital) -> appends to both Sheet1 & Check In const regRow = [ new Date(), payload.name, String(payload.whatsapp || ''), payload.email || '', payload.charmCount, payload.charmPrice, payload.invitingFriend, payload.referredBy || '', payload.inviteShareUrl || '' ]; const regSheet = ss.getSheetByName(REG_SHEET_NAME); regSheet.appendRow(regRow); regSheet.getRange(regSheet.getLastRow(), COL.WA).setNumberFormat('@').setValue(String(payload.whatsapp || '')); const ci = getOrCreateCheckInSheet_(ss); ci.appendRow(regRow.concat(['Unpaid', '', '', 'No', '', ''])); setTextFormats_(ci, ci.getLastRow()); ci.getRange(ci.getLastRow(), COL.WA).setValue(String(payload.whatsapp || '')); return ContentService.createTextOutput(JSON.stringify({ status: 'ok', target: REG_SHEET_NAME })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
